  <script src="https://cdnjs.cloudflare.com/ajax/libs/PapaParse/5.4.1/papaparse.min.js"></script>

  <script>

    let activities = [];

    let gradeLevels = new Set();

    let subjects = new Set();


    document.getElementById('csvFile').addEventListener('change', function(e) {

      const file = e.target.files[0];

      if (file) {

        Papa.parse(file, {

          header: true,

          skipEmptyLines: true,

          complete: function(results) {

            activities = results.data.filter(row => row['Grade Level'] && row['Subject']);

            gradeLevels.clear(); subjects.clear();

            activities.forEach(a => { gradeLevels.add(a['Grade Level']); subjects.add(a['Subject']); });

            populateFilters();

            document.getElementById('ideaBox').classList.add('hidden');

          }

        });

      }

    });


    function populateFilters() {

      const gradeSelect = document.getElementById('gradeFilter');

      gradeSelect.innerHTML = '<option value="">All</option>' +

        Array.from(gradeLevels).sort((a, b) => a.localeCompare(b)).map(g => `<option value="${g}">${g}</option>`).join('');

      const subjectSelect = document.getElementById('subjectFilter');

      subjectSelect.innerHTML = '<option value="">All</option>' +

        Array.from(subjects).sort((a, b) => a.localeCompare(b)).map(s => `<option value="${s}">${s}</option>`).join('');

    }


    function generateIdea() {

      let gradeVal = document.getElementById('gradeFilter').value;

      let subjectVal = document.getElementById('subjectFilter').value;

      let filtered = activities.filter(a =>

        (!gradeVal || a['Grade Level'] === gradeVal) &&

        (!subjectVal || a['Subject'] === subjectVal)

      );

      if (filtered.length === 0) {

        document.getElementById('ideaBox').innerHTML = 'No ideas found for selected filters.';

        document.getElementById('ideaBox').classList.remove('hidden');

        return;

      }

      const idea = filtered[Math.floor(Math.random() * filtered.length)];

      document.getElementById('ideaBox').innerHTML = `

        <div><span class="field">Grade:</span> ${idea['Grade Level']}</div>

        <div><span class="field">Subject:</span> ${idea['Subject']}</div>

        <div><span class="field">Activity Title:</span> ${idea['Activity Title']
